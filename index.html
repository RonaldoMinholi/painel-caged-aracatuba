name: Importar Novo CAGED on: schedule: - cron: '20 10 * * *' workflow_dispatch: inputs: competencia: description: 'AAAAMM. Vazio = procura os últimos 4 meses.' required: false importar_tabela_oficial: description: 'Importar a Tabela 8.1 oficial com histórico e ajustes.' type: boolean required: false default: false jobs: import: runs-on: ubuntu-latest permissions: contents: read steps: - name: Baixar código do projeto uses: actions/checkout@v4 - name: Preparar Python uses: actions/setup-python@v5 with: python-version: '3.12' - name: Instalar dependências run: pip install -r importer/requirements.txt - name: Importar Tabela 8.1 oficial if: ${{ inputs.importar_tabela_oficial == true }} env: SUPABASE_URL: ${{ secrets.SUPABASE_URL }} SUPABASE_SERVICE_ROLE_KEY: ${{ secrets.SUPABASE_SERVICE_ROLE_KEY }} TABLES_FOLDER_URL: https://drive.google.com/drive/folders/1SLFCZ184KseNP8W9Xc6YMe6rCrHeuaiU?usp=drive_link SOURCE_PAGE_URL: https://www.gov.br/trabalho-e-emprego/pt-br/acesso-a-informacao/acoes-e-programas/programas-projetos-acoes-obras-e-atividades/estatisticas-trabalho/novo-caged/2026/julho/pagina-inicial run: | mkdir -p official-tables gdown --folder "$TABLES_FOLDER_URL" --output official-tables --remaining-ok workbook=$(find official-tables -type f -iname '*.xlsx' | head -n 1) test -n "$workbook" python importer/import_official_tables.py \ --file "$workbook" \ --source-url "$SOURCE_PAGE_URL" - name: Importar microdados da competência if: ${{ inputs.importar_tabela_oficial != true }} env: SUPABASE_URL: ${{ secrets.SUPABASE_URL }} SUPABASE_SERVICE_ROLE_KEY: ${{ secrets.SUPABASE_SERVICE_ROLE_KEY }} CAGED_SOURCE_BASE_URL: ${{ secrets.CAGED_SOURCE_BASE_URL }} COMPETENCIA: ${{ inputs.competencia }} run: | if [ -n "$COMPETENCIA" ]; then python importer/import_caged.py --competencia "$COMPETENCIA" else for offset in 1 2 3 4; do competencia=$(date -d "$(date +%Y-%m-01) - ${offset} month" +%Y%m) python importer/import_caged.py --competencia "$competencia" && break || true done fi
